% notation_table.tex
% Table of notation used throughout the manuscript.
%
% Pulled into the appendix by appendix.tex. Keep one row per symbol, and keep
% the rows in the order the symbols are introduced in the body -- a reader
% scanning for a symbol they just met reads top-down.
%
% Add a row as:  $\symbol$ & Description of the symbol \\

\begin{table}[htbp]
\centering
\caption{Notation}
\label{tab:notation}
\begin{threeparttable}
\begin{tabular}{ll}
\toprule
Symbol       & Description \\
\midrule
$TAC(t)$     & Transdermal alcohol concentration at time $t$ \\
$I(t)$       & Alcohol ingestion (intake) rate at time $t$ \\
$M(t)$       & Alcohol metabolism (elimination) rate at time $t$ \\
$S(t)$       & Two-state switching process; $1$ within a drinking episode, $0$ otherwise \\
$q_{01}$     & Transition rate for entering a drinking episode \\
$q_{10}$     & Transition rate for exiting a drinking episode \\
$k$          & Natural frequency of the damped oscillator, $k > 0$ \\
$\zeta$      & Damping ratio, $\zeta > 1$ \\
$\eta$       & Scaling factor on the input function $I(t)$, $\eta > 0$ \\
$x_{it|t-1}$ & One-step-ahead predicted latent state for person $i$ at time $t$ \\
$P_{it|t-1}$ & One-step-ahead predicted state covariance for person $i$ at time $t$ \\
$y_{it}$     & Observed measurement for person $i$ at time $t$ \\
$H$          & Measurement (loading) matrix mapping latent states to observations \\
$v_{it}$     & Prediction error (innovation), $y_{it} - H x_{it|t-1}$ \\
$F_{it}$     & Prediction error variance, $H P_{it|t-1} H' + \texttt{mnoise}$ \\
$A$          & Continuous-time drift matrix \\
$Q_c$        & Continuous-time process noise covariance \\
$A_d, Q_d$   & Discrete-time drift and process noise over the interval $\Delta t$ \\
$\Delta t$   & Interval between successive measurement occasions \\
\bottomrule
\end{tabular}
\begin{tablenotes}[flushleft]
\small
\item \textit{Note.} Subscript $i$ indexes person and subscript $t$ indexes
measurement occasion throughout.
\end{tablenotes}
\end{threeparttable}
\end{table}
